/** คืน {y,m,d} จาก 'YYYY-MM-DD' (เลขล้วน ไม่พึ่ง timezone) */
function parse(iso: string): { y: number; m: number; d: number } {
  const [y, m, d] = iso.split('-').map(Number);
  return { y, m, d };
}

function pad(n: number): string {
  return String(n).padStart(2, '0');
}

/** วันเริ่ม + N ปี − 1 วัน → 'YYYY-MM-DD' (ใช้ UTC เพื่อความแน่นอน) */
export function addYearsMinusDay(startIso: string, years: number): string {
  const { y, m, d } = parse(startIso);
  const t = Date.UTC(y + years, m - 1, d);
  const end = new Date(t - 24 * 60 * 60 * 1000);
  return `${end.getUTCFullYear()}-${pad(end.getUTCMonth() + 1)}-${pad(end.getUTCDate())}`;
}

/** 'YYYY-MM-DD' → 'YYYY-MM' */
export function monthOf(iso: string): string {
  return iso.slice(0, 7);
}

/** ไล่เดือนต่อเนื่อง count เดือนจาก startMonth ('YYYY-MM') */
export function listMonths(startMonth: string, count: number): string[] {
  const [y, m] = startMonth.split('-').map(Number);
  const out: string[] = [];
  for (let i = 0; i < count; i++) {
    const total = (m - 1) + i;
    const yy = y + Math.floor(total / 12);
    const mm = (total % 12) + 1;
    out.push(`${yy}-${pad(mm)}`);
  }
  return out;
}
```

import type { Tier, TierRule } from './types';

/** มูลค่าต่อปี = มูลค่าสัญญา ÷ จำนวนปี (SPEC §7.1) */
export function annualValue(valueTotal: number, termYears: number): number {
  return valueTotal / termYears;
}

/** เลือก tier ที่ minAnnual <= annual สูงสุด (เกณฑ์มาจาก tier_rules เท่านั้น ห้าม hardcode) */
export function tierOf(annual: number, rules: TierRule[]): Tier {
  const sorted = [...rules].sort((a, b) => a.minAnnual - b.minAnnual);
  let picked: Tier = sorted[0].tier;
  for (const r of sorted) {
    if (annual >= r.minAnnual) picked = r.tier;
    else break;
  }
  return picked;
}
```

import type { Contract, Visit } from './types';
import { serviceMonths } from './serviceMonths';
import { monthOf } from './date';

/**
 * ตกบริการเมื่อ: month อยู่ในเดือนบริการ และสัญญาไม่ถูกยกเลิก (ก่อน/ในเดือนนั้น)
 * และไม่มี visit ประเภท routine ในเดือน month (SPEC §7.4)
 * หมายเหตุ: ผู้เรียกควรจำกัดเฉพาะเดือนที่ผ่านไปแล้ว
 */
export function isMissedService(contract: Contract, month: string, visits: Visit[]): boolean {
  if (!serviceMonths(contract.startDate, contract.termYears).includes(month)) return false;
  if (contract.cancelledAt && monthOf(contract.cancelledAt) <= month) return false;
  const hasRoutine = visits.some(
    (v) => v.contractId === contract.id && v.visitType === 'routine' && monthOf(v.visitDate) === month,
  );
  return !hasRoutine;
}
```

import { serviceMonths } from './serviceMonths';
import { monthOf } from './date';

/** สร้างแถว snapshot รูปแบบ Dashboard (SPEC §7.6, §9.1) สำหรับ 1 สาขา × 1 เดือน */
export function buildBranchSnapshot(
  branch: BranchRef,
  contracts: Contract[],
  month: string,
  rules: TierRule[],
): SnapshotRow {
  const row: SnapshotRow = {
    area: branch.area, branch: branch.branch,
    s1: 0, s3: 0, m1: 0, m3: 0, l1: 0, l3: 0, xl1: 0, xl3: 0,
    cust: 0, vS: 0, vM: 0, vL: 0, vXL: 0, value: 0, value_annual: 0,
  };

  for (const c of contracts) {
    if (!serviceMonths(c.startDate, c.termYears).includes(month)) continue;
    if (c.cancelledAt && monthOf(c.cancelledAt) <= month) continue;

    const av = annualValue(c.valueTotal, c.termYears);
    const tier = tierOf(av, rules);
    const countKey = `${tier.toLowerCase()}${c.termYears}` as keyof SnapshotRow;
    (row[countKey] as number) += 1;

    const valueKey = `v${tier}` as 'vS' | 'vM' | 'vL' | 'vXL';
    row[valueKey] += c.valueTotal;

    row.cust += 1;
    row.value += c.valueTotal;
    row.value_annual += av;
  }
  return row;
}
```
